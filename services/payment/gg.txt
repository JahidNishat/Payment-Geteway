Question 1
Using the same OrderService, write a test for the failure case:

email sending fails with error "smtp connection refused"
PlaceOrder should return that error
Follow the ARRANGE → ACT → ASSERT pattern.

type FakeEmailSender struct{
    err error
    to string
    from string
    called bool
}

func(f *FakeEmailSender) Send(to, from string) error {
    f.called = true
    f.to = to
    return f.err
}

func TestPlaceOrder_SendsEmail(t *testing.T) {
    // ARRANGE
    fake := &FakeEmailSender{err: errors.New("smtp connection refused)}
    svc := &OrderService{emailer: fake}
    
    // ACT
    err := svc.PlaceOrder("nishat@example.com")
    
    // ASSERT
    if err == nil {
        t.Fatalf("expected error, but got nil")
    }
    if !fake.called {
        t.Fatal("expected email to be sent, but Send was never called")
    }
    if fake.calledWith != "nishat@example.com" {
        t.Errorf("sent to %s, want nishat@example.com", fake.calledWith)
    }
}
